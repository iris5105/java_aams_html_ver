  ```javascript
  function getFilterCorpGr() {
      var el = currentPane.querySelector('#corpGrSelect') || document.querySelector('#corpGrSelect');
      if (el && el.value) return el.value;

      var matchSaved = document.cookie.match(/(?:^|;\s*)savedCorpGr=([^;]*)/);
      if (matchSaved && matchSaved[1]) return decodeURIComponent(matchSaved[1]);

      var matchCorp = document.cookie.match(/(?:^|;\s*)corpGr=([^;]*)/);
      if (matchCorp && matchCorp[1]) return decodeURIComponent(matchCorp[1]);

      if (serverCorpGr) return serverCorpGr;
      if (typeof g_corp_gr !== 'undefined' && g_corp_gr) return g_corp_gr;
      return '';
  }
  ```

- **그리드 행 선택 삼중 안전망 (Triple Fallback, 규칙 8)**:
  ```javascript
  setupTabulatorRowSelection(grid, function(row, data) {

  grid.on("rowClick", function(e, row) {
      syncDetail(row, row ? row.getData() : null);
  });
  grid.on("rowSelectionChanged", function(data, rows) {
      if (rows && rows.length > 0) syncDetail(rows[0], rows[0].getData());

  ```

- **상세 리포트 동기화 및 줌 120% 적용**:
  ```javascript
  function syncDetail(row, masterRowData) {
      if (!row) return;
      if (!row.isSelected || !row.isSelected()) row.select();
      var data = masterRowData || (row.getData ? row.getData() : null);
      if (!data) return;

      var currentKey = corpGr + "_" + ymd + "_" + (data.fundCd || "");
      if (lastLoadedMasterKey === currentKey) return;
      lastLoadedMasterKey = currentKey;

      loadReportPreview(data);
  }
  ```

### 3.3 매퍼 SQL 계층 (`Uzm0hyMapper.xml`)

```xml
<select id="selectUzm0hyList" resultType="com.kfp.aams.domain.daily.dto.Uzm0hyDto">
    SELECT t1.corp_gr                                       AS corpGr
         , t1.enc_acct_no                                   AS encAcctNo
         , t1.fund_cd                                       AS fundCd
         , t1.fund_nm                                       AS fundNm
         , t1.type_gb                                       AS typeGb
         , TO_CHAR(t1.fst_seolj_ymd, 'YYYY-MM-DD')          AS fstSeoljYmd
         , t1.sintak_gigan                                  AS sintakGigan
         , TO_CHAR(t1.bf_gyul_ymd, 'YYYY-MM-DD')           AS bfGyulYmd
         , TO_CHAR(t1.af_gyul_ymd, 'YYYY-MM-DD')           AS afGyulYmd
         , t1.target_jasan                                  AS targetJasan
         , t1.gyul_gi                                       AS gyulGi
         , t1.haeji_gb                                      AS haejiGb
         , TO_CHAR(t1.haeji_ymd, 'YYYY-MM-DD')             AS haejiYmd
         , t1.re_seolj_year                                 AS reSeoljYear
         , t1.re_seolj_aek                                  AS reSeoljAek
         , t1.mg_cd                                         AS mgCd
         , TO_CHAR(t1.re_seolj_ymd, 'YYYY-MM-DD')          AS reSeoljYmd
         , t1.series_gb                                     AS seriesGb
         , TO_DECRYPTS(t1.enc_acct_no)                     AS acctNo
         , se.gugan                                         AS gugan
      FROM SZM0IA t1
         , SZX0SE se
     WHERE t1.corp_gr                     = #{corpGr}
       AND NVL(t1.haeji_ymd, TO_DATE('2099-12-31', 'YYYY-MM-DD')) > TO_DATE(REPLACE(#{ymd}, '-', ''), 'YYYYMMDD')
       AND se.corp_gr                     = t1.corp_gr
       AND se.series_gb                   = t1.series_gb
     ORDER BY t1.haeji_gb ASC, t1.mg_cd ASC, t1.fund_nm ASC, t1.re_seolj_ymd ASC
</select>
```
