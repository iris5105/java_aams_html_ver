## 3. 계층별 세부 코드 및 구성 분석

### 3.1 화면 및 클라이언트 스크립트 ([w_ja010m3.html](file:///d:/work/java_aams_html_ver/src/main/resources/templates/views/daily/w_ja010m3.html))

```javascript
function onRowSelected(data) {
    if (!data || !data.mrdNm) {
        if (frame) frame.style.display = "none";
        if (placeholder) placeholder.style.display = "block";
        currentReportUrl = "";
        return;
    }

    const mrd = data.mrdNm.trim();
    const params = `fund_cd[${data.fundCd}] gyul_ymd[${data.gyulYmd.replace(/-/g, '.')}]`;
    const rawUrl = `/api/rd/pdf?mrd=${encodeURIComponent(mrd)}&param=${encodeURIComponent(params)}`;
    
    // AamsReport 공통 유틸을 통한 120% 줌 일괄 적용
    const url = (window.AamsReport && typeof window.AamsReport.formatPreviewUrl === 'function')
        ? window.AamsReport.formatPreviewUrl(rawUrl)
        : rawUrl + "#toolbar=1&navpanes=0&zoom=120";
    
    currentReportUrl = url;

    if (placeholder) placeholder.style.display = "none";
    if (frame) {
        frame.style.display = "block";
        frame.src = url;
    }
}
```

### 3.2 매퍼 SQL 계층 (`Ja010m3Mapper.xml`)

```xml
<select id="selectSettlementList" resultType="com.kfp.aams.domain.daily.dto.Ja010m3Dto">
    SELECT A.CORP_GR     AS corpGr,
           A.FUND_CD     AS fundCd,
           B.FUND_NM     AS fundNm,
           A.GYUL_YMD    AS gyulYmd,
           A.MRD_NM      AS mrdNm,
           A.CONFIRM_YN  AS confirmYn
      FROM FW_GYUL_MST A
      JOIN SZX0AA B ON A.CORP_GR = B.CORP_GR AND A.FUND_CD = B.FUND_CD
     WHERE A.CORP_GR = #{corpGr}
       AND A.GYUL_YMD = #{ymd}
     ORDER BY A.FUND_CD ASC
</select>
```
